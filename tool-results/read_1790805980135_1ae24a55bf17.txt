    47→class Burst {
    48→  points: THREE.Points
    49→  private vel: Float32Array
    50→  private t = 0
    51→  private life: number
    52→
    53→  constructor(scene: THREE.Scene, pos: THREE.Vector3, color: number, count = 14, speed = 3, life = 0.7, size = 0.14) {
    54→    this.life = life
    55→    const positions = new Float32Array(count * 3)
    56→    this.vel = new Float32Array(count * 3)
    57→    for (let i = 0; i < count; i++) {
    58→      positions[i * 3] = pos.x
    59→      positions[i * 3 + 1] = pos.y
    60→      positions[i * 3 + 2] = pos.z
    61→      const a = Math.random() * Math.PI * 2
    62→      const s = speed * (0.4 + Math.random() * 0.6)
    63→      this.vel[i * 3] = Math.cos(a) * s
    64→      this.vel[i * 3 + 1] = 1.5 + Math.random() * 2.5
    65→      this.vel[i * 3 + 2] = Math.sin(a) * s
    66→    }
    67→    const geo = new THREE.BufferGeometry()
    68→    geo.setAttribute('position', new THREE.BufferAttribute(positions, 3))
    69→    const mat = new THREE.PointsMaterial({
    70→      color, size, transparent: true, opacity: 1, depthWrite: false,
    71→    })
    72→    this.points = new THREE.Points(geo, mat)
    73→    scene.add(this.points)
    74→  }
    75→
    76→  update(dt: number, scene: THREE.Scene): boolean {
    77→    this.t += dt
    78→    const pos = this.points.geometry.getAttribute('position') as THREE.BufferAttribute
    79→    for (let i = 0; i < pos.count; i++) {
    80→      pos.setXYZ(
    81→        i,
    82→        pos.getX(i) + this.vel[i * 3] * dt,
    83→        pos.getY(i) + this.vel[i * 3 + 1] * dt,
    84→        pos.getZ(i) + this.vel[i * 3 + 2] * dt
    85→      )
    86→      this.vel[i * 3 + 1] -= 5 * dt
    87→    }
    88→    pos.needsUpdate = true
    89→    const mat = this.points.material as THREE.PointsMaterial
    90→    mat.opacity = Math.max(0, 1 - this.t / this.life)
    91→    if (this.t >= this.life) {
    92→      scene.remove(this.points)
    93→      this.points.geometry.dispose()
    94→      mat.dispose()
    95→      return false
    96→    }
    97→    return true
    98→  }
    99→}

   101→class FloatText {
   102→  sprite: THREE.Sprite
   103→  t = 0
   104→
   105→  constructor(text: string, color: string, pos: THREE.Vector3) {
   106→    const canvas = document.createElement('canvas')
   107→    canvas.width = 256
   108→    canvas.height = 96
   109→    const ctx = canvas.getContext('2d')!
   110→    ctx.font = 'bold 54px monospace'
   111→    ctx.textAlign = 'center'
   112→    ctx.textBaseline = 'middle'
   113→    ctx.lineWidth = 10
   114→    ctx.strokeStyle = 'rgba(0,0,0,0.9)'
   115→    ctx.strokeText(text, 128, 48)
   116→    ctx.fillStyle = color
   117→    ctx.fillText(text, 128, 48)
   118→    const tex = new THREE.CanvasTexture(canvas)
   119→    tex.colorSpace = THREE.SRGBColorSpace
   120→    const mat = new THREE.SpriteMaterial({ map: tex, transparent: true, depthWrite: false })
   121→    this.sprite = new THREE.Sprite(mat)
   122→    this.sprite.scale.set(2.4, 0.9, 1)
   123→    this.sprite.position.copy(pos)
   124→  }
   125→
   126→  update(dt: number, scene: THREE.Scene): boolean {
   127→    this.t += dt
   128→    this.sprite.position.y += dt * 1.4
   129→    const mat = this.sprite.material as THREE.SpriteMaterial
   130→    mat.opacity = Math.max(0, 1 - this.t / 1.1)
   131→    if (this.t >= 1.1) {
   132→      scene.remove(this.sprite)
   133→      mat.map?.dispose()
   134→      mat.dispose()
   135→      return false
   136→    }
   137→    return true
   138→  }
   139→}

   334→  rest() {

   354→  levelUp(stat: 'vit' | 'end' | 'str') {
   355→    if (this.phase !== 'rest') return
   356→    const cost = this.nextCost()
   357→    if (this.player.souls < cost) return
   358→    this.player.souls -= cost
   359→    this.player.applyLevel(stat)
   360→    this.save()
   361→    this.sfx.levelUp()
   362→    this.emit(true)
   363→  }

   365→  interact() {
   366→    if (this.phase !== 'playing') return
   367→    // bloodstain
   368→    if (this.bloodstain && this.bloodstain.mesh.position.distanceTo(this.player.pos) < 1.7) {
   369→      this.player.souls += this.bloodstain.amount
   370→      this.spawnBurst(this.bloodstain.mesh.position, 0x59ff6a, 22, 3.5)
   371→      this.spawnText(`+${this.bloodstain.amount}`, '#59ff6a', this.player.pos.clone().add(new THREE.Vector3(0, 2.2, 0)))
   372→      this.engine.scene.remove(this.bloodstain.mesh)
   373→      this.bloodstain = null
   374→      this.sfx.souls()
   375→      this.emit(true)
   376→      return
   377→    }
   378→    // bonfire
   379→    const bPos = new THREE.Vector3(BONFIRE.x, this.world.surfaceAt(BONFIRE.x, BONFIRE.z), BONFIRE.z)
   380→    if (bPos.distanceTo(this.player.pos) < 2.6) {
   381→      this.rest()
   382→      return
   383→    }
   384→    // fog gate
   385→    if (!this.bossActive && !this.bossFell && this.player.pos.z < GATE_Z + 3.2 && this.player.pos.z > GATE_Z - 1) {
   386→      this.fogPassT = 0.75
   387→      this.sfx.bossRoar()
   388→    }
   389→  }
   390→
   391→  toggleLock() {
   392→    if (this.player.lockedTarget) {
   393→      this.player.lockedTarget = null
   394→      return
   395→    }
   396→    const camF = new THREE.Vector3(-Math.sin(this.camYaw), 0, -Math.cos(this.camYaw))
   397→    let best: Enemy | null = null
   398→    let bestScore = Infinity
   399→    for (const e of this.allEnemies) {
   400→      if (!e.alive || (e.isBoss && !this.bossActive)) continue
   401→      const to = e.pos.clone().sub(this.player.pos)
   402→      const d = to.length()
   403→      if (d > 15) continue
   404→      to.normalize()
   405→      const dot = to.dot(camF)
   406→      if (dot < 0.1) continue
   407→      const score = d * (2 - dot)
   408→      if (score < bestScore) {
   409→        bestScore = score
   410→        best = e
   411→      }
   412→    }
   413→    if (best) {
   414→      this.player.lockedTarget = { pos: best.pos, alive: true, isBoss: best.isBoss }
   415→    }
   416→  }

   440→  private save() {
   441→    try {
   442→      const data: SaveData = {
   443→        souls: this.player.souls,
   444→        level: this.player.level,
   445→        vit: this.player.vit,
   446→        end: this.player.end,
   447→        str: this.player.str,
   448→      }
   449→      localStorage.setItem(SAVE_KEY, JSON.stringify(data))
   450→    } catch { /* ignore */ }

   453→  private loadSave() {
   454→    try {
   455→      const raw = localStorage.getItem(SAVE_KEY)
   456→      if (!raw) return
   457→      const d = JSON.parse(raw) as SaveData
   458→      // re-apply levels from scratch for consistency
   459→      const target = { vit: d.vit ?? 0, end: d.end ?? 0, str: d.str ?? 0 }
   460→      this.player.vit = 0
   461→      this.player.end = 0
   462→      this.player.str = 0
   463→      this.player.maxHp = 95
   464→      this.player.maxStamina = 95
   465→      for (let i = 0; i < target.vit; i++) this.player.applyLevel('vit')
   466→      for (let i = 0; i < target.end; i++) this.player.applyLevel('end')
   467→      for (let i = 0; i < target.str; i++) this.player.applyLevel('str')
   468→      this.player.level = 1 + target.vit + target.end + target.str
   469→      this.player.souls = d.souls ?? 0
   470→    } catch { /* ignore */ }
   471→  }

   473→  private playerStrike(def: PlayerStrikeDef) {
   474→    const fwd = new THREE.Vector3(Math.sin(this.player.yaw), 0, Math.cos(this.player.yaw))
   475→    let hits = 0
   476→    for (const e of this.allEnemies) {
   477→      if (!e.alive) continue
   478→      const to = e.pos.clone().sub(this.player.pos)
   479→      const d = to.length()
   480→      if (d > def.range + (e.isBoss ? 1.2 : 0.3)) continue
   481→      to.y = 0
   482→      to.normalize()
   483→      if (to.dot(fwd) < Math.cos(def.arc)) continue
   484→      const dmg = Math.max(1, Math.round(def.dmg * (0.92 + Math.random() * 0.16)))
   485→      e.takeDamage(dmg, this, this.player.pos.x, this.player.pos.z)
   486→      hits++
   487→    }
   488→    if (hits > 0) {
   489→      this.hitstop = def.heavy ? 0.09 : 0.06
   490→      if (def.heavy) {
   491→        this.sfx.heavy()
   492→        this.shake = Math.max(this.shake, 0.18)
   493→      } else {
   494→        this.sfx.hit()
   495→      }
   496→      // spark burst at first victim
   497→      const victim = this.allEnemies.find((e) => e.alive && e.pos.distanceTo(this.player.pos) < def.range + 1.4)
   498→      if (victim) this.spawnBurst(victim.pos.clone().add(new THREE.Vector3(0, 1.3, 0)), 0xffe08a, 8, 2.2, 0.4)
   499→    }
   500→  }

   619→      if (p.z < GATE_Z + 0.9) p.z = GATE_Z + 0.9
   620→    }

   622→    if (this.bossActiveBarrier && !this.bossFell) {
   623→      p.x = Math.max(BOSS_CENTER.x - 7.6, Math.min(BOSS_CENTER.x + 7.6, p.x))

   640→  private loop(rawDt: number) {
   641→    // periodic shadow map refresh (big perf win)
   642→    this.shadowTimer += rawDt
   643→    if (this.shadowTimer > 0.15) {
   644→      this.shadowTimer = 0
   645→      this.sun.shadow.needsUpdate = true
   646→    }
   647→
   648→    // hitstop slow-mo
   649→    let dt = rawDt
   650→    if (this.hitstop > 0) {
   651→      this.hitstop -= rawDt
   652→      dt = rawDt * 0.08
   653→    }
   654→    this.time += dt
   655→
   656→    const input = this.engine.input
   657→
   658→    if (this.phase === 'menu') {
   659→      // slow orbit around bonfire
   660→      this.camYaw += dt * 0.12
   661→      this.world.update(dt)
   662→      this.updateBonfire(dt)
   663→      this.player.h.group.position.copy(this.player.pos)
   664→      this.player.h.group.rotation.y = this.player.yaw
   665→      this.updateCameraMenu(dt)
   666→      this.emit(false)
   667→      return
   668→    }
   669→
   670→    if (this.phase === 'dead') {
   671→      this.deadT += rawDt
   672→      this.player.update({ input, camYaw: this.camYaw, dt, world: this.world, game: this })
   673→      this.world.update(dt)
   674→      this.updateBonfire(dt)
   675→      this.updateEffects(dt)
   676→      if (this.deadT > 2.9) {
   677→        this.deadT = 0
   678→        this.respawn()
   679→      }
   680→      this.updateCameraFollow(dt, true)
   681→      this.emit(false)
   682→      return
   683→    }
   684→
   685→    if (this.phase === 'rest') {
   686→      this.world.update(dt)
   687→      this.updateBonfire(dt)
   688→      this.updateEffects(dt)
   689→      this.updateCameraFollow(dt, false)
   690→      this.emit(false)
   691→      return
   692→    }
   693→
   694→    /* ---- playing ---- */
   695→
   696→    // camera input
   697→    const { dx, dy } = input.takeMouse()
   698→    if (dx !== 0 || dy !== 0) this.lockLastMove = 0
   699→    else this.lockLastMove += dt
   700→    const sens = 0.0031
   701→    this.camYaw -= dx * sens
   702→    this.camPitch += dy * sens
   703→    this.camPitch = Math.max(-0.45, Math.min(1.15, this.camPitch))
   704→    const wheel = input.takeWheel()
   705→    if (wheel !== 0) this.camDistTarget = Math.max(3.4, Math.min(8.5, this.camDistTarget + wheel * 0.004))
   706→    this.camDist += (this.camDistTarget - this.camDist) * Math.min(1, 8 * dt)
   707→
   708→    // global keys
   709→    if (input.consume('KeyQ')) this.toggleLock()
   710→    if (input.consume('KeyF')) this.interact()
   711→    if (input.consume('Escape')) {
   712→      this.player.lockedTarget = null
   713→      input.releaseLock()
   714→    }
   715→
   716→    // fog pass animation
   717→    if (this.fogPassT > 0) {
   718→      this.fogPassT -= dt
   719→      this.player.pos.z -= 5.5 * dt
   720→      // swing the camera around to face the boss arena cinematically
   721→      let dYaw = -this.camYaw
   722→      while (dYaw > Math.PI) dYaw -= Math.PI * 2
   723→      while (dYaw < -Math.PI) dYaw += Math.PI * 2
   724→      this.camYaw += dYaw * Math.min(1, 5 * dt)
   725→      this.camPitch += (0.32 - this.camPitch) * Math.min(1, 4 * dt)
   726→      if (this.fogPassT <= 0) {
   727→        this.bossActive = true
   728→        this.bossActiveBarrier = true
   729→        this.boss.active = true
   730→        this.sfx.bossRoar()
   731→        this.shake = Math.max(this.shake, 0.4)
   732→      }
   733→    }
   734→
   735→    // lock-on validity + soft camera pull
   736→    if (this.player.lockedTarget) {
   737→      const lt = this.player.lockedTarget
   738→      const e = this.allEnemies.find((en) => en.pos === lt.pos)
   739→      if (!e || e.dead) {
   740→        this.player.lockedTarget = null
   741→      } else if (e.pos.distanceTo(this.player.pos) > 17) {
   742→        this.player.lockedTarget = null
   743→      } else if (this.lockLastMove > 0.8) {
   744→        const desired = e.yaw + Math.PI
   745→        let d = desired - this.camYaw
   746→        while (d > Math.PI) d -= Math.PI * 2
   747→        while (d < -Math.PI) d += Math.PI * 2
   748→        this.camYaw += d * Math.min(1, 2.2 * dt)
   749→      }
   750→    }
   751→
   752→    this.clampPlayer()
   753→    this.player.update({ input, camYaw: this.camYaw, dt, world: this.world, game: this })
   754→
   755→    // death transition
   756→    if (this.player.state === 'dead') {
   757→      this.phase = 'dead'
   758→      this.deadT = 0
   759→      this.spawnBloodstain()
   760→      this.player.lockedTarget = null
   761→      input.releaseLock()
   762→      this.sfx.death()
   763→      this.emit(true)
   764→      return
   765→    }
   766→
   767→    for (const e of this.enemies) e.update(dt, this.player, this)
   768→    if (!this.bossFell) this.boss.update(dt, this.player, this)
   769→
   770→    this.world.update(dt)
   771→    this.updateBonfire(dt)
   772→    this.updateEffects(dt)
   773→    this.updateOrbs(dt)
   774→
   775→    // prompt
   776→    this.prompt = this.fogPassT > 0 ? null : this.detectPrompt()
   777→    if (this.prompt && input.isTouch && input.consume('TouchInteract')) this.interact()
   778→
   779→    // banner timer
   780→    if (this.banner) {
   781→      this.bannerT += rawDt
   782→      if (this.banner === 'bossfell' && this.bannerT > 3.4) {
   783→        this.banner = null
   784→        this.emit(true)
   785→      }
   786→    }
   787→
   788→    // hurt vignette decay
   789→    this.hurtFlash = Math.max(0, this.hurtFlash - rawDt * 1.6)
   790→    const lowHp = this.player.hp / this.player.maxHp < 0.25 && this.player.alive
   791→    const vig = Math.max(this.hurtFlash, lowHp ? 0.22 + Math.sin(this.time * 5) * 0.08 : 0)
   792→    this.vignette.style.opacity = String(vig)
   793→
   794→    this.updateCameraFollow(dt, false)
   795→    this.updateReticle()
   796→    this.emit(false)
   797→  }

   799→  private updateBonfire(dt: number) {
   800→    this.bonfireLight.intensity = 2.6 + Math.sin(this.time * 13) * 0.35 + Math.random() * 0.3
   801→    const pos = this.bonfireFlame.geometry.getAttribute('position') as THREE.BufferAttribute
   802→    const bY = this.world.surfaceAt(BONFIRE.x, BONFIRE.z)
   803→    for (let i = 0; i < pos.count; i++) {
   804→      const seed = this.flameSeeds[i]
   805→      const cycle = (this.time * (0.8 + seed * 0.7) + seed * 3) % 1
   806→      const ang = seed * Math.PI * 2 + this.time * (0.5 + seed)
   807→      const r = 0.28 * (1 - cycle)
   808→      pos.setXYZ(
   809→        i,
   810→        BONFIRE.x + Math.cos(ang) * r,
   811→        bY + 0.15 + cycle * 1.7,
   812→        BONFIRE.z + Math.sin(ang) * r
   813→      )
   814→    }
   815→    pos.needsUpdate = true
   816→    void dt
   817→  }

   819→  private updateEffects(dt: number) {
   820→    this.bursts = this.bursts.filter((b) => b.update(dt, this.engine.scene))
   821→    this.texts = this.texts.filter((t) => t.update(dt, this.engine.scene))
   822→    // bloodstain bob
   823→    if (this.bloodstain) {
   824→      this.bloodstain.mesh.rotation.y += dt * 1.5
   825→      this.bloodstain.mesh.position.y += Math.sin(this.time * 3) * dt * 0.12
   826→    }
   827→  }

   829→  private updateOrbs(dt: number) {
   830→    this.orbs = this.orbs.filter((o) => {
   831→      o.t += dt / 0.7
   832→      const target = this.player.pos.clone().add(new THREE.Vector3(0, 1.1, 0))
   833→      o.mesh.position.lerpVectors(o.from, target, Math.min(1, o.t))
   834→      o.mesh.rotation.y += dt * 6
   835→      if (o.t >= 1) {
   836→        this.engine.scene.remove(o.mesh)
   837→        this.player.souls += o.amount
   838→        this.sfx.souls()
   839→        this.spawnBurst(target, 0x59ff6a, 10, 2)
   840→        return false
   841→      }
   842→      return true
   843→    })
   844→  }

   846→  private updateCameraMenu(dt: number) {
   847→    const bY = this.world.surfaceAt(BONFIRE.x, BONFIRE.z)

   849→    const d = 6.2
   850→    this.camPos.set(
   851→      this.camTarget.x + Math.sin(this.camYaw) * d,
   852→      this.camTarget.y + 2.2,
   853→      this.camTarget.z + Math.cos(this.camYaw) * d
   854→    )
   855→    this.engine.camera.position.lerp(this.camPos, Math.min(1, 3 * dt))
   856→    this.engine.camera.lookAt(this.camTarget)
   857→  }

   859→  private updateCameraFollow(dt: number, dead: boolean) {
   860→    const p = this.player
   861→    this.camTarget.set(p.pos.x, p.pos.y + 1.5, p.pos.z)
   862→    const dist = dead ? this.camDist + 2.5 : this.camDist
   863→    const cp = Math.max(0.12, this.camPitch)
   864→    let cx = this.camTarget.x + Math.sin(this.camYaw) * Math.cos(cp) * dist
   865→    let cz = this.camTarget.z + Math.cos(this.camYaw) * Math.cos(cp) * dist
   866→    let cy = this.camTarget.y + Math.sin(cp) * dist
   867→    // keep above ground
   868→    const ground = this.world.surfaceAt(cx, cz) + 0.45
   869→    if (cy < ground) cy = ground
   870→    this.camPos.set(cx, cy, cz)
   871→    this.engine.camera.position.lerp(this.camPos, Math.min(1, 11 * dt))
   872→    if (this.shake > 0) {
   873→      this.shake = Math.max(0, this.shake - dt * 1.8)
   874→      const s = this.shake * 0.25
   875→      this.engine.camera.position.x += (Math.random() - 0.5) * s
   876→      this.engine.camera.position.y += (Math.random() - 0.5) * s
   877→    }
   878→    this.engine.camera.lookAt(this.camTarget)
   879→  }

   881→  private updateReticle() {
   882→    const lt = this.player.lockedTarget
   883→    if (!lt || !lt.alive || this.phase !== 'playing') {
   884→      this.reticle.style.display = 'none'
   885→      return
   886→    }
   887→    const v = lt.pos.clone().add(new THREE.Vector3(0, lt.isBoss ? 3.6 : 1.7, 0))
   888→    v.project(this.engine.camera)
   889→    if (v.z > 1) {
   890→      this.reticle.style.display = 'none'
   891→      return
   892→    }
   893→    const w = this.container.clientWidth
   894→    const h = this.container.clientHeight
   895→    this.reticle.style.display = 'block'
   896→    this.reticle.style.left = `${(v.x * 0.5 + 0.5) * w}px`
   897→    this.reticle.style.top = `${(-v.y * 0.5 + 0.5) * h}px`
   898→  }

   900→  private emit(force: boolean) {
   901→    if (!this.onState) return
   902→    const p = this.player
   903→    const bossVisible = this.bossActive && !this.bossFell
   904→    const s: HudState = {
   905→      phase: this.phase,
   906→      hp: Math.max(0, Math.round(p.hp)),
   907→      maxHp: p.maxHp,
   908→      st: Math.round(p.stamina),
   909→      maxSt: p.maxStamina,
   910→      souls: Math.floor(p.souls),
   911→      level: p.level,
   912→      estus: p.estus,
   913→      maxEstus: p.maxEstus,
   914→      vit: p.vit,
   915→      end: p.end,
   916→      str: p.str,
   917→      nextCost: this.nextCost(),
   918→      bossName: bossVisible ? this.boss.name : null,
   919→      bossHp: Math.max(0, Math.round(this.boss.hp)),
   920→      bossMax: this.boss.maxHp,
   921→      prompt: this.phase === 'playing' ? this.prompt : null,
   922→      banner: this.phase === 'dead' ? 'died' : this.banner,
   923→    }
   924→    const json = JSON.stringify(s)
   925→    if (force || json !== this.lastHudJson) {
   926→      this.lastHudJson = json
   927→      this.onState(s)
   928→    }
   929→  }
