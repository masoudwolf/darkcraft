    48→class Burst {
    49→  points: THREE.Points
    50→  private vel: Float32Array
    51→  private t = 0
    52→  private life: number
    53→
    54→  constructor(scene: THREE.Scene, pos: THREE.Vector3, color: number, count = 14, speed = 3, life = 0.7, size = 0.14) {
    55→    this.life = life
    56→    const positions = new Float32Array(count * 3)
    57→    this.vel = new Float32Array(count * 3)
    58→    for (let i = 0; i < count; i++) {
    59→      positions[i * 3] = pos.x
    60→      positions[i * 3 + 1] = pos.y
    61→      positions[i * 3 + 2] = pos.z
    62→      const a = Math.random() * Math.PI * 2
    63→      const s = speed * (0.4 + Math.random() * 0.6)
    64→      this.vel[i * 3] = Math.cos(a) * s
    65→      this.vel[i * 3 + 1] = 1.5 + Math.random() * 2.5
    66→      this.vel[i * 3 + 2] = Math.sin(a) * s
    67→    }
    68→    const geo = new THREE.BufferGeometry()
    69→    geo.setAttribute('position', new THREE.BufferAttribute(positions, 3))
    70→    const mat = new THREE.PointsMaterial({
    71→      color, size, transparent: true, opacity: 1, depthWrite: false,
    72→    })
    73→    this.points = new THREE.Points(geo, mat)
    74→    scene.add(this.points)
    75→  }
    76→
    77→  update(dt: number, scene: THREE.Scene): boolean {
    78→    this.t += dt
    79→    const pos = this.points.geometry.getAttribute('position') as THREE.BufferAttribute
    80→    for (let i = 0; i < pos.count; i++) {
    81→      pos.setXYZ(
    82→        i,
    83→        pos.getX(i) + this.vel[i * 3] * dt,
    84→        pos.getY(i) + this.vel[i * 3 + 1] * dt,
    85→        pos.getZ(i) + this.vel[i * 3 + 2] * dt
    86→      )
    87→      this.vel[i * 3 + 1] -= 5 * dt
    88→    }
    89→    pos.needsUpdate = true
    90→    const mat = this.points.material as THREE.PointsMaterial
    91→    mat.opacity = Math.max(0, 1 - this.t / this.life)
    92→    if (this.t >= this.life) {
    93→      scene.remove(this.points)
    94→      this.points.geometry.dispose()
    95→      mat.dispose()
    96→      return false
    97→    }
    98→    return true
    99→  }
   100→}

   102→class FloatText {
   103→  sprite: THREE.Sprite
   104→  t = 0
   105→
   106→  constructor(text: string, color: string, pos: THREE.Vector3) {
   107→    const canvas = document.createElement('canvas')
   108→    canvas.width = 256
   109→    canvas.height = 96
   110→    const ctx = canvas.getContext('2d')!
   111→    ctx.font = 'bold 54px monospace'
   112→    ctx.textAlign = 'center'
   113→    ctx.textBaseline = 'middle'
   114→    ctx.lineWidth = 10
   115→    ctx.strokeStyle = 'rgba(0,0,0,0.9)'
   116→    ctx.strokeText(text, 128, 48)
   117→    ctx.fillStyle = color
   118→    ctx.fillText(text, 128, 48)
   119→    const tex = new THREE.CanvasTexture(canvas)
   120→    tex.colorSpace = THREE.SRGBColorSpace
   121→    const mat = new THREE.SpriteMaterial({ map: tex, transparent: true, depthWrite: false })
   122→    this.sprite = new THREE.Sprite(mat)
   123→    this.sprite.scale.set(2.4, 0.9, 1)
   124→    this.sprite.position.copy(pos)
   125→  }
   126→
   127→  update(dt: number, scene: THREE.Scene): boolean {
   128→    this.t += dt
   129→    this.sprite.position.y += dt * 1.4
   130→    const mat = this.sprite.material as THREE.SpriteMaterial
   131→    mat.opacity = Math.max(0, 1 - this.t / 1.1)
   132→    if (this.t >= 1.1) {
   133→      scene.remove(this.sprite)
   134→      mat.map?.dispose()
   135→      mat.dispose()
   136→      return false
   137→    }
   138→    return true
   139→  }

   405→  toggleLock() {
   406→    if (this.player.lockedTarget) {
   407→      this.player.lockedTarget = null
   408→      return
   409→    }
   410→    const camF = new THREE.Vector3(-Math.sin(this.camYaw), 0, -Math.cos(this.camYaw))
   411→    let best: Enemy | null = null
   412→    let bestScore = Infinity
   413→    for (const e of this.allEnemies) {
   414→      if (!e.alive || (e.isBoss && !this.bossActive)) continue
   415→      const to = e.pos.clone().sub(this.player.pos)
   416→      const d = to.length()
   417→      if (d > 15) continue
   418→      to.normalize()
   419→      const dot = to.dot(camF)
   420→      if (dot < 0.1) continue
   421→      const score = d * (2 - dot)
   422→      if (score < bestScore) {
   423→        bestScore = score
   424→        best = e
   425→      }
   426→    }
   427→    if (best) {
   428→      this.player.lockedTarget = { pos: best.pos, alive: true, isBoss: best.isBoss }
   429→    }
   430→  }

   470→  private loadSave() {
   471→    try {
   472→      const raw = localStorage.getItem(SAVE_KEY)
   473→      if (!raw) return
   474→      const d = JSON.parse(raw) as SaveData
   475→      // re-apply levels from scratch for consistency
   476→      const target = { vit: d.vit ?? 0, end: d.end ?? 0, str: d.str ?? 0 }
   477→      this.player.vit = 0
   478→      this.player.end = 0
   479→      this.player.str = 0
   480→      this.player.maxHp = 95
   481→      this.player.maxStamina = 95
   482→      for (let i = 0; i < target.vit; i++) this.player.applyLevel('vit')
   483→      for (let i = 0; i < target.end; i++) this.player.applyLevel('end')
   484→      for (let i = 0; i < target.str; i++) this.player.applyLevel('str')
   485→      this.player.level = 1 + target.vit + target.end + target.str
   486→      this.player.souls = d.souls ?? 0
   487→    } catch { /* ignore */ }
   488→  }

   490→  private playerStrike(def: PlayerStrikeDef) {
   491→    const fwd = new THREE.Vector3(Math.sin(this.player.yaw), 0, Math.cos(this.player.yaw))
   492→    let hits = 0
   493→    for (const e of this.allEnemies) {
   494→      if (!e.alive) continue
   495→      const to = e.pos.clone().sub(this.player.pos)
   496→      const d = to.length()
   497→      if (d > def.range + (e.isBoss ? 1.2 : 0.3)) continue
   498→      to.y = 0
   499→      to.normalize()
   500→      if (to.dot(fwd) < Math.cos(def.arc)) continue
   501→      const dmg = Math.max(1, Math.round(def.dmg * (0.92 + Math.random() * 0.16)))
   502→      e.takeDamage(dmg, this, this.player.pos.x, this.player.pos.z)
   503→      hits++
   504→    }
   505→    if (hits > 0) {
   506→      this.hitstop = def.heavy ? 0.09 : 0.06
   507→      if (def.heavy) {
   508→        this.sfx.heavy()
   509→        this.shake = Math.max(this.shake, 0.18)
   510→      } else {
   511→        this.sfx.hit()
   512→      }
   513→      // spark burst at first victim
   514→      const victim = this.allEnemies.find((e) => e.alive && e.pos.distanceTo(this.player.pos) < def.range + 1.4)
   515→      if (victim) this.spawnBurst(victim.pos.clone().add(new THREE.Vector3(0, 1.3, 0)), 0xffe08a, 8, 2.2, 0.4)
   516→    }
   517→  }

   689→  private loop(rawDt: number) {
   690→    // periodic shadow map refresh (big perf win)
   691→    this.shadowTimer += rawDt
   692→    if (this.shadowTimer > 0.15) {
   693→      this.shadowTimer = 0
   694→      this.sun.shadow.needsUpdate = true
   695→    }
   696→
   697→    // hitstop slow-mo
   698→    let dt = rawDt
   699→    if (this.hitstop > 0) {
   700→      this.hitstop -= rawDt
   701→      dt = rawDt * 0.08
   702→    }
   703→    this.time += dt
   704→
   705→    const input = this.engine.input
   706→
   707→    if (this.phase === 'menu') {
   708→      // slow orbit around bonfire
   709→      this.camYaw += dt * 0.12
   710→      this.world.update(dt)
   711→      this.updateBonfire(dt)
   712→      this.player.h.group.position.copy(this.player.pos)
   713→      this.player.h.group.rotation.y = this.player.yaw
   714→      this.updateCameraMenu(dt)
   715→      this.emit(false)
   716→      return
   717→    }
   718→
   719→    if (this.phase === 'dead') {
   720→      this.deadT += rawDt
   721→      this.player.update({ input, camYaw: this.camYaw, dt, world: this.world, game: this })
   722→      this.world.update(dt)
   723→      this.updateBonfire(dt)
   724→      this.updateEffects(dt)
   725→      if (this.deadT > 2.9) {
   726→        this.deadT = 0
   727→        this.respawn()
   728→      }
   729→      this.updateCameraFollow(dt, true)
   730→      this.emit(false)
   731→      return
   732→    }
   733→
   734→    if (this.phase === 'rest') {
   735→      this.world.update(dt)
   736→      this.updateBonfire(dt)
   737→      this.updateEffects(dt)
   738→      this.updateCameraFollow(dt, false)
   739→      this.emit(false)
   740→      return
   741→    }
   742→
   743→    /* ---- playing ---- */
   744→
   745→    // camera input
   746→    const { dx, dy } = input.takeMouse()
   747→    if (dx !== 0 || dy !== 0) this.lockLastMove = 0
   748→    else this.lockLastMove += dt
   749→    const sens = 0.0031
   750→    this.camYaw -= dx * sens
   751→    this.camPitch += dy * sens
   752→    this.camPitch = Math.max(-0.45, Math.min(1.15, this.camPitch))
   753→    const wheel = input.takeWheel()
   754→    if (wheel !== 0) this.camDistTarget = Math.max(3.4, Math.min(8.5, this.camDistTarget + wheel * 0.004))
   755→    this.camDist += (this.camDistTarget - this.camDist) * Math.min(1, 8 * dt)
   756→
   757→    // global keys
   758→    if (input.consume('KeyQ')) this.toggleLock()
   759→    if (input.consume('KeyF')) this.interact()
   760→    if (input.consume('Escape')) {
   761→      this.player.lockedTarget = null
   762→      input.releaseLock()
   763→    }
   764→
   765→    // fog pass animation
   766→    if (this.fogPassT > 0) {
   767→      this.fogPassT -= dt
   768→      this.player.pos.z -= 5.5 * dt
   769→      // swing the camera around to face the boss arena cinematically
   770→      let dYaw = -this.camYaw
   771→      while (dYaw > Math.PI) dYaw -= Math.PI * 2
   772→      while (dYaw < -Math.PI) dYaw += Math.PI * 2
   773→      this.camYaw += dYaw * Math.min(1, 5 * dt)
   774→      this.camPitch += (0.32 - this.camPitch) * Math.min(1, 4 * dt)
   775→      if (this.fogPassT <= 0) {
   776→        this.bossActive = true
   777→        this.bossActiveBarrier = true
   778→        this.boss.active = true
   779→        this.sfx.bossRoar()
   780→        this.shake = Math.max(this.shake, 0.4)
   781→      }
   782→    }
   783→
   784→    // lock-on validity + soft camera pull
   785→    if (this.player.lockedTarget) {
   786→      const lt = this.player.lockedTarget
   787→      const e = this.allEnemies.find((en) => en.pos === lt.pos)
   788→      if (!e || e.dead) {
   789→        this.player.lockedTarget = null
   790→      } else if (e.pos.distanceTo(this.player.pos) > 17) {
   791→        this.player.lockedTarget = null
   792→      } else if (this.lockLastMove > 0.8) {
   793→        const desired = e.yaw + Math.PI
   794→        let d = desired - this.camYaw
   795→        while (d > Math.PI) d -= Math.PI * 2
   796→        while (d < -Math.PI) d += Math.PI * 2
   797→        this.camYaw += d * Math.min(1, 2.2 * dt)
   798→      }
   799→    }
   800→
   801→    this.clampPlayer()
   802→    this.player.update({ input, camYaw: this.camYaw, dt, world: this.world, game: this })
   803→
   804→    // death transition
   805→    if (this.player.state === 'dead') {
   806→      this.phase = 'dead'
   807→      this.deadT = 0
   808→      this.spawnBloodstain()
   809→      this.player.lockedTarget = null
   810→      input.releaseLock()
   811→      this.sfx.death()
   812→      this.emit(true)
   813→      return
   814→    }
   815→
   816→    for (const e of this.enemies) e.update(dt, this.player, this)
   817→    if (!this.bossFell) this.boss.update(dt, this.player, this)
   818→
   819→    this.world.update(dt)
   820→    this.updateBonfire(dt)
   821→    this.updateEffects(dt)
   822→    this.updateOrbs(dt)
   823→
   824→    // prompt
   825→    this.prompt = this.fogPassT > 0 ? null : this.detectPrompt()
   826→    if (this.prompt && input.isTouch && input.consume('TouchInteract')) this.interact()
   827→
   828→    // banner timer
   829→    if (this.banner) {
   830→      this.bannerT += rawDt
   831→      if (this.banner === 'bossfell' && this.bannerT > 3.4) {
   832→        this.banner = null
   833→        this.emit(true)
   834→      }
   835→    }
   836→
   837→    // hurt vignette decay
   838→    this.hurtFlash = Math.max(0, this.hurtFlash - rawDt * 1.6)
   839→    const lowHp = this.player.hp / this.player.maxHp < 0.25 && this.player.alive
   840→    const vig = Math.max(this.hurtFlash, lowHp ? 0.22 + Math.sin(this.time * 5) * 0.08 : 0)
   841→    this.vignette.style.opacity = String(vig)
   842→
   843→    this.updateCameraFollow(dt, false)
   844→    this.updateReticle()
   845→    this.emit(false)
   846→  }

   848→  private updateBonfire(dt: number) {
   849→    this.bonfireLight.intensity = 2.6 + Math.sin(this.time * 13) * 0.35 + Math.random() * 0.3
   850→    const pos = this.bonfireFlame.geometry.getAttribute('position') as THREE.BufferAttribute
   851→    const bY = this.world.surfaceAt(BONFIRE.x, BONFIRE.z)
   852→    for (let i = 0; i < pos.count; i++) {
   853→      const seed = this.flameSeeds[i]
   854→      const cycle = (this.time * (0.8 + seed * 0.7) + seed * 3) % 1
   855→      const ang = seed * Math.PI * 2 + this.time * (0.5 + seed)
   856→      const r = 0.28 * (1 - cycle)
   857→      pos.setXYZ(
   858→        i,
   859→        BONFIRE.x + Math.cos(ang) * r,
   860→        bY + 0.15 + cycle * 1.7,
   861→        BONFIRE.z + Math.sin(ang) * r
   862→      )
   863→    }
   864→    pos.needsUpdate = true
   865→    void dt
   866→  }

   868→  private updateEffects(dt: number) {
   869→    this.bursts = this.bursts.filter((b) => b.update(dt, this.engine.scene))
   870→    this.texts = this.texts.filter((t) => t.update(dt, this.engine.scene))
   871→    // creeper explosion flash lights
   872→    this.boomLights = this.boomLights.filter((b) => {
   873→      b.t += dt
   874→      b.light.intensity = Math.max(0, 7 * (1 - b.t / 0.4))
   875→      if (b.t >= 0.4) {
   876→        this.engine.scene.remove(b.light)
   877→        b.light.dispose()
   878→        return false
   879→      }
   880→      return true
   881→    })
   882→    // bloodstain bob
   883→    if (this.bloodstain) {
   884→      this.bloodstain.mesh.rotation.y += dt * 1.5
   885→      this.bloodstain.mesh.position.y += Math.sin(this.time * 3) * dt * 0.12
   886→    }
   887→  }

   889→  private updateOrbs(dt: number) {
   890→    this.orbs = this.orbs.filter((o) => {
   891→      o.t += dt / 0.7
   892→      const target = this.player.pos.clone().add(new THREE.Vector3(0, 1.1, 0))
   893→      o.mesh.position.lerpVectors(o.from, target, Math.min(1, o.t))
   894→      o.mesh.rotation.y += dt * 6
   895→      if (o.t >= 1) {
   896→        this.engine.scene.remove(o.mesh)
   897→        this.player.souls += o.amount
   898→        this.sfx.souls()
   899→        this.spawnBurst(target, 0x59ff6a, 10, 2)
   900→        return false
   901→      }
   902→      return true
   903→    })
   904→  }

   906→  private updateCameraMenu(dt: number) {
   907→    const bY = this.world.surfaceAt(BONFIRE.x, BONFIRE.z)

   911→      this.camTarget.x + Math.sin(this.camYaw) * d,
   912→      this.camTarget.y + 2.2,
   913→      this.camTarget.z + Math.cos(this.camYaw) * d
   914→    )
   915→    this.engine.camera.position.lerp(this.camPos, Math.min(1, 3 * dt))
   916→    this.engine.camera.lookAt(this.camTarget)

   919→  private updateCameraFollow(dt: number, dead: boolean) {
   920→    const p = this.player
   921→    this.camTarget.set(p.pos.x, p.pos.y + 1.5, p.pos.z)
   922→    const dist = dead ? this.camDist + 2.5 : this.camDist
   923→    const cp = Math.max(0.12, this.camPitch)
   924→    let cx = this.camTarget.x + Math.sin(this.camYaw) * Math.cos(cp) * dist
   925→    let cz = this.camTarget.z + Math.cos(this.camYaw) * Math.cos(cp) * dist
   926→    let cy = this.camTarget.y + Math.sin(cp) * dist
   927→    // keep above ground
   928→    const ground = this.world.surfaceAt(cx, cz) + 0.45
   929→    if (cy < ground) cy = ground
   930→    this.camPos.set(cx, cy, cz)
   931→    this.engine.camera.position.lerp(this.camPos, Math.min(1, 11 * dt))
   932→    if (this.shake > 0) {
   933→      this.shake = Math.max(0, this.shake - dt * 1.8)
   934→      const s = this.shake * 0.25
   935→      this.engine.camera.position.x += (Math.random() - 0.5) * s
   936→      this.engine.camera.position.y += (Math.random() - 0.5) * s
   937→    }
   938→    this.engine.camera.lookAt(this.camTarget)
   939→  }

   941→  private updateReticle() {
   942→    const lt = this.player.lockedTarget
   943→    if (!lt || !lt.alive || this.phase !== 'playing') {
   944→      this.reticle.style.display = 'none'
   945→      return
   946→    }
   947→    const v = lt.pos.clone().add(new THREE.Vector3(0, lt.isBoss ? 3.6 : 1.7, 0))
   948→    v.project(this.engine.camera)
   949→    if (v.z > 1) {
   950→      this.reticle.style.display = 'none'
   951→      return
   952→    }
   953→    const w = this.container.clientWidth
   954→    const h = this.container.clientHeight
   955→    this.reticle.style.display = 'block'
   956→    this.reticle.style.left = `${(v.x * 0.5 + 0.5) * w}px`
   957→    this.reticle.style.top = `${(-v.y * 0.5 + 0.5) * h}px`
   958→  }

   960→  private emit(force: boolean) {
   961→    if (!this.onState) return
   962→    const p = this.player
   963→    const bossVisible = this.bossActive && !this.bossFell
   964→    const s: HudState = {
   965→      phase: this.phase,
   966→      hp: Math.max(0, Math.round(p.hp)),
   967→      maxHp: p.maxHp,
   968→      st: Math.round(p.stamina),
   969→      maxSt: p.maxStamina,
   970→      souls: Math.floor(p.souls),
   971→      level: p.level,
   972→      estus: p.estus,
   973→      maxEstus: p.maxEstus,
   974→      vit: p.vit,
   975→      end: p.end,
   976→      str: p.str,
   977→      nextCost: this.nextCost(),
   978→      bossName: bossVisible ? this.boss.name : null,
   979→      bossHp: Math.max(0, Math.round(this.boss.hp)),
   980→      bossMax: this.boss.maxHp,
   981→      prompt: this.phase === 'playing' ? this.prompt : null,
   982→      banner: this.phase === 'dead' ? 'died' : this.banner,
   983→      blocking: p.state === 'block',
   984→    }
   985→    const json = JSON.stringify(s)
   986→    if (force || json !== this.lastHudJson) {
   987→      this.lastHudJson = json
   988→      this.onState(s)
   989→    }
   990→  }
